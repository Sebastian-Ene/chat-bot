**Version A — dispatch table for content blocks (least branching)**

Replaces the `if/elif` chain in `format_content` with a lookup table, so adding a new block type later is a one-line addition instead of another branch.

```python
BLOCK_FORMATTERS = {
    "text": lambda b: b.get("text", ""),
    "thinking": lambda b: f"_(thinking)_\n{b.get('thinking', '')}",
    "tool_use": lambda b: (
        f"**Tool call: `{b.get('name')}`**\n```json\n"
        f"{json.dumps(b.get('input', {}), indent=2)}\n```"
    ),
    "tool_result": lambda b: (
        f"**Tool result**\n```\n"
        f"{b['content'] if isinstance(b.get('content'), str) else json.dumps(b.get('content'), indent=2)}\n```"
    ),
}


def format_content(content: str | list[dict]) -> str:
    if isinstance(content, str):
        return content
    fmt = lambda b: BLOCK_FORMATTERS.get(b["type"], lambda b: f"```json\n{json.dumps(b, indent=2)}\n```")(b)
    return "\n\n".join(fmt(block) for block in content)
```
